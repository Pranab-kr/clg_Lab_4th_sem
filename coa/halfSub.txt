module half_subtractor(
    input A,
    input B,
    output Difference,
    output Borrow
);

assign Difference = A ^ B;
assign Borrow = (~A) & B;

endmodule

//simu

module half_subtractor_tb;

reg A, B;
wire Difference, Borrow;

half_subtractor uut(
    .A(A),
    .B(B),
    .Difference(Difference),
    .Borrow(Borrow)
);

initial begin
    A=0; B=0;
    #10 A=0; B=1;
    #10 A=1; B=0;
    #10 A=1; B=1;
    #10;
end

endmodule
