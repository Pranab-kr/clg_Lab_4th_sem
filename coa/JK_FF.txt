module jk_ff(
    input wire clk,
    input wire rst,
    input wire j,
    input wire k,
    output reg q
);

always @(posedge clk or posedge rst)
begin

    if(rst)
        q <= 1'b0;

    else begin

        if(j == 1'b0 && k == 1'b0)
            q <= q;

        else if(j == 1'b0 && k == 1'b1)
            q <= 1'b0;

        else if(j == 1'b1 && k == 1'b0)
            q <= 1'b1;

        else
            q <= ~q;

    end

end

endmodule

//simu

module jk_ff_tb;

reg clk;
reg rst;
reg j;
reg k;

wire q;

jk_ff uut(
    .clk(clk),
    .rst(rst),
    .j(j),
    .k(k),
    .q(q)
);

always #5 clk = ~clk;

initial begin

    clk = 0;
    rst = 1;
    j = 0;
    k = 0;

    #10;
    rst = 0;

    #10 j = 1; k = 0;
    #20 j = 0; k = 1;
    #20 j = 0; k = 0;
    #20 j = 1; k = 1;

    #20;
    $finish;

end

endmodule
