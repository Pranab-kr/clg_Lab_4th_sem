module decoder_2to4(
    input EN,
    input [1:0] A,
    output reg [3:0] Y
);

always @(*) begin

    if(EN) begin

        case(A)
            2'b00: Y = 4'b0001;
            2'b01: Y = 4'b0010;
            2'b10: Y = 4'b0100;
            2'b11: Y = 4'b1000;
            default: Y = 4'b0000;
        endcase

    end

    else begin
        Y = 4'b0000;
    end

end

endmodule

//simu

module decoder_2to4_tb;

reg EN;
reg [1:0] A;
wire [3:0] Y;

decoder_2to4 uut(
    .EN(EN),
    .A(A),
    .Y(Y)
);

initial begin

    EN = 0; A = 2'b00;

    #10 EN = 1; A = 2'b00;
    #10 A = 2'b01;
    #10 A = 2'b10;
    #10 A = 2'b11;

    #10 EN = 0;

    #10;
    $finish;

end

endmodule
