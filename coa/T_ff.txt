module t_ff(
    input wire clk,
    input wire rst,
    input wire t,
    output reg q
);

always @(posedge clk or posedge rst)
begin

    if(rst)
        q <= 1'b0;

    else if(t)
        q <= ~q;

    else
        q <= q;

end

endmodule

//simu

module t_ff_tb;

reg clk;
reg rst;
reg t;

wire q;

t_ff uut(
    .clk(clk),
    .rst(rst),
    .t(t),
    .q(q)
);

always #5 clk = ~clk;

initial begin

    clk = 0;
    rst = 1;
    t = 0;

    #10;
    rst = 0;

    #10 t = 1;
    #20 t = 0;
    #20 t = 1;
    #20 t = 1;
    #20 t = 0;

    #20;
    $finish;

end

endmodule
