module sr_ff(
    input wire clk,
    input wire rst,
    input wire s,
    input wire r,
    output reg q
);

always @(posedge clk or posedge rst)
begin

    if(rst)
        q <= 1'b0;

    else begin

        if(s == 1'b0 && r == 1'b0)
            q <= q;

        else if(s == 1'b0 && r == 1'b1)
            q <= 1'b0;

        else if(s == 1'b1 && r == 1'b0)
            q <= 1'b1;

        else
            q <= 1'bx;

    end

end

endmodule

module sr_ff_tb;

reg clk;
reg rst;
reg s;
reg r;

wire q;

sr_ff uut(
    .clk(clk),
    .rst(rst),
    .s(s),
    .r(r),
    .q(q)
);

always #5 clk = ~clk;

initial begin

    clk = 0;
    rst = 1;
    s = 0;
    r = 0;

    #10;
    rst = 0;

    #10 s = 1; r = 0;
    #20 s = 0; r = 1;
    #20 s = 0; r = 0;
    #20 s = 1; r = 1;

    #20;
    $finish;

end

endmodule
