module t_to_d(

    input wire clk,
    input wire rst,
    input wire d,
    output reg q

);

wire t;

assign t = d ^ q;

always @(posedge clk or posedge rst)
begin

    if(rst)
        q <= 1'b0;

    else begin

        if(t)
            q <= ~q;

        else
            q <= q;

    end

end

endmodule

module t_to_d_tb;

reg clk;
reg rst;
reg d;

wire q;

t_to_d uut(

    .clk(clk),
    .rst(rst),
    .d(d),
    .q(q)

);

always #5 clk = ~clk;

initial begin

    clk = 0;
    rst = 1;
    d = 0;

    #10 rst = 0;

    #10 d = 1;
    #20 d = 0;
    #20 d = 1;
    #20 d = 1;
    #20 d = 0;

    #20;
    $finish;

end

endmodule
