module mux_8to1(
    input EN,
    input [7:0] I,
    input [2:0] S,
    output reg Y
);

always @(*) begin

    if(EN) begin

        case(S)
            3'b000: Y = I[0];
            3'b001: Y = I[1];
            3'b010: Y = I[2];
            3'b011: Y = I[3];
            3'b100: Y = I[4];
            3'b101: Y = I[5];
            3'b110: Y = I[6];
            3'b111: Y = I[7];
            default: Y = 1'b0;
        endcase

    end

    else begin
        Y = 1'b0;
    end

end

endmodule

module mux_8to1_tb;

reg EN;
reg [7:0] I;
reg [2:0] S;
wire Y;

mux_8to1 uut(
    .EN(EN),
    .I(I),
    .S(S),
    .Y(Y)
);

initial begin

    I = 8'b10101010;

    EN = 0; S = 3'b000;

    #10 EN = 1; S = 3'b000;
    #10 S = 3'b001;
    #10 S = 3'b010;
    #10 S = 3'b011;
    #10 S = 3'b100;
    #10 S = 3'b101;
    #10 S = 3'b110;
    #10 S = 3'b111;

    #10 EN = 0;

    #10;
    $finish;

end

endmodule
