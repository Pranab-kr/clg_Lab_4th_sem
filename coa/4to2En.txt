module encoder_4to2(
    input EN,
    input [3:0] D,
    output reg [1:0] Y
);

always @(*) begin

    if(EN) begin

        case(D)
            4'b0001: Y = 2'b00;
            4'b0010: Y = 2'b01;
            4'b0100: Y = 2'b10;
            4'b1000: Y = 2'b11;
            default: Y = 2'b00;
        endcase

    end

    else begin
        Y = 2'b00;
    end

end

endmodule

module encoder_4to2_tb;

reg EN;
reg [3:0] D;
wire [1:0] Y;

encoder_4to2 uut(
    .EN(EN),
    .D(D),
    .Y(Y)
);

initial begin

    EN = 0; D = 4'b0001;

    #10 EN = 1; D = 4'b0001;
    #10 D = 4'b0010;
    #10 D = 4'b0100;
    #10 D = 4'b1000;

    #10 EN = 0;

    #10;
    $finish;

end

endmodule
