module sr_to_jk(

    input wire clk,
    input wire rst,
    input wire j,
    input wire k,
    output reg q

);

wire s, r;

assign s = j & ~q;
assign r = k & q;

always @(posedge clk or posedge rst)
begin

    if(rst)
        q <= 1'b0;

    else begin

        case({s,r})

            2'b00: q <= q;
            2'b01: q <= 1'b0;
            2'b10: q <= 1'b1;
            2'b11: q <= 1'bx;

        endcase

    end

end

endmodule

//simu

module sr_to_jk_tb;

reg clk;
reg rst;
reg j;
reg k;

wire q;

sr_to_jk uut(

    .clk(clk),
    .rst(rst),
    .j(j),
    .k(k),
    .q(q)

);

always #5 clk = ~clk;

initial begin

    clk = 0;
    rst = 1;
    j = 0;
    k = 0;

    #10 rst = 0;

    #10 j = 1; k = 0;
    #20 j = 0; k = 0;
    #20 j = 0; k = 1;
    #20 j = 1; k = 1;
    #20 j = 0; k = 0;

    #20;
    $finish;

end

endmodule
