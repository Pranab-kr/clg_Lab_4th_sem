module xor_gate(
    input A,
    input B,
    output Y
);

assign Y = A ^ B;

endmodule

//simu

module xor_gate_tb;

reg A, B;
wire Y;

xor_gate uut(
    .A(A),
    .B(B),
    .Y(Y)
);

initial begin
    A=0; B=0;
    #10 A=0; B=1;
    #10 A=1; B=0;
    #10 A=1; B=1;
    #10;
end

endmodule
