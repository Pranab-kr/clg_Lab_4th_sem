module d_ff(
    input wire clk,
    input wire rst,
    input wire d,
    output reg q
);

always @(posedge clk or posedge rst)
begin

    if(rst)
        q <= 1'b0;

    else
        q <= d;

end

endmodule

//simu

module d_ff_tb;

reg clk;
reg rst;
reg d;

wire q;

d_ff uut(
    .clk(clk),
    .rst(rst),
    .d(d),
    .q(q)
);

always #5 clk = ~clk;

initial begin

    clk = 0;
    rst = 1;
    d = 0;

    #10;
    rst = 0;

    #10 d = 1;
    #20 d = 0;
    #20 d = 1;
    #20 d = 0;

    #20;
    $finish;

end

endmodule
