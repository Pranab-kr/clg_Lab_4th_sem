module comp1bit(
    input A,
    input B,
    output G,
    output L,
    output E
);

assign G = A & (~B);
assign L = (~A) & B;
assign E = ~(A ^ B);

endmodule

//simu

module comp1bit_tb;

reg A, B;
wire G, L, E;

comp1bit uut(
    .A(A),
    .B(B),
    .G(G),
    .L(L),
    .E(E)
);

initial begin

    A = 0; B = 0;
    #10 A = 0; B = 1;
    #10 A = 1; B = 0;
    #10 A = 1; B = 1;

    #10;
    $finish;

end

endmodule
