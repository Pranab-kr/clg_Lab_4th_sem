module mux_4to1(
    input EN,
    input [3:0] I,
    input [1:0] S,
    output reg Y
);

always @(*) begin

    if(EN) begin

        case(S)
            2'b00: Y = I[0];
            2'b01: Y = I[1];
            2'b10: Y = I[2];
            2'b11: Y = I[3];
            default: Y = 1'b0;
        endcase

    end

    else begin
        Y = 1'b0;
    end

end

endmodule

//simu

module mux_4to1_tb;

reg EN;
reg [3:0] I;
reg [1:0] S;
wire Y;

mux_4to1 uut(
    .EN(EN),
    .I(I),
    .S(S),
    .Y(Y)
);

initial begin

    I = 4'b1010;

    EN = 0; S = 2'b00;

    #10 EN = 1; S = 2'b00;
    #10 S = 2'b01;
    #10 S = 2'b10;
    #10 S = 2'b11;

    #10 EN = 0;

    #10;
    $finish;

end

endmodule
