module full_subtractor(
    input A,
    input B,
    input Bin,
    output Difference,
    output Bout
);

assign Difference = A ^ B ^ Bin;
assign Bout = (~A & B) | (~A & Bin) | (B & Bin);

endmodule

//simu

module full_subtractor_tb;

reg A, B, Bin;
wire Difference, Bout;

full_subtractor uut(
    .A(A),
    .B(B),
    .Bin(Bin),
    .Difference(Difference),
    .Bout(Bout)
);
